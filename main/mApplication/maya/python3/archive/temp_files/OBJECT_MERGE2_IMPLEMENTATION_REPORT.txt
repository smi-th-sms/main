Core Sync Function:
-------------------
```python
def sync_object_merge2(proxy_node=None):
    # Get proxy and merge nodes
    proxy_node = proxy_node or hou.node("/obj/assets/Proxy")
    merge_node = hou.node("/obj/assets/Cache/object_merge2")
    
    # Get cloth parts count
    cloth_parts_count = proxy_node.parm("cloth_parts").eval()
    
    # Update numobj
    merge_node.parm("numobj").set(cloth_parts_count)
    
    # Update paths
    for i in range(1, cloth_parts_count + 1):
        part_name = proxy_node.parm(f"part_name_{i}").eval()
        out_path = f"../Proxy/geo_{part_name}/OUT_{part_name}_PROXY"
        merge_node.parm(f"objpath{i}").set(out_path)
```
